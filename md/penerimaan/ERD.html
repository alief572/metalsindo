Data Flow - Save Penerimaan

save_penerimaan() :
|
+-- INSERT tr_invoice_payment (1 row header)
|
+-- LOOP detail rows:
|   +-- IF type = 'invoice':
|   |   +-- INSERT tr_invoice_payment_detail (total_bayar_idr = positif)
|   |   +-- UPDATE tr_invoice SET sisa_invoice_idr -= X, total_bayar_idr += X
|   |   +-- UPDATE tr_spk_marketing SET total_bayar_so += X
|   |
|   +-- IF type = 'cn':
|       +-- INSERT tr_invoice_payment_detail (total_bayar_idr = negatif)
|       +-- INSERT tr_cn_cross (amount_crossed = abs(X))
|
+-- IF tambah_lebih_bayar > 0:
|   +-- INSERT tr_unlocated_bank (saldo = tambah_lebih_bayar)
|
+-- IF id_unlocated != '':
    +-- UPDATE tr_unlocated_bank SET saldo -= X

CN Balance Calculation

CN_Balance = SUM(dt_returpenjualan.total_harga) WHERE id_retur = ?
           - COALESCE(SUM(tr_cn_cross.amount_crossed), 0) WHERE id_retur = ?

Validasi:
  [OK] crossing_amount > 0
  [OK] crossing_amount <= CN_Balance
  [NO] CN_Balance = 0 maka CN sudah fully crossed (tidak muncul di list)
